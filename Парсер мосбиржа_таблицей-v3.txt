proxy_url = urllib.request.getproxies().get("https")

if not proxy_url:
    raise RuntimeError("Корпоративный HTTPS-прокси не найден")


ps_script = r"""
$ErrorActionPreference = "Stop";

$proxy = [System.Net.WebProxy]::new($env:MOEX_PROXY);
$proxy.Credentials = [System.Net.CredentialCache]::DefaultNetworkCredentials;

$client = [System.Net.WebClient]::new();
$client.Proxy = $proxy;

try {
    $bytes = $client.DownloadData($env:MOEX_URL);

    [Console]::Out.Write(
        [Convert]::ToBase64String($bytes)
    );
}
finally {
    $client.Dispose();
}
"""


# =====================================================
# 3. ЗАГРУЗКА JSON ЧЕРЕЗ ПРОКСИ БЕЗ ФАЙЛОВ
# =====================================================

def download_moex_json(url):

    env = os.environ.copy()

    env["MOEX_PROXY"] = proxy_url
    env["MOEX_URL"] = url

    result = subprocess.run(
        [
            "powershell.exe",
            "-NoProfile",
            "-NonInteractive",
            "-Command",
            ps_script
        ],
        capture_output=True,
        text=True,
        errors="replace",
        env=env,
        timeout=90
    )

    if result.returncode != 0:
        raise RuntimeError(
            f"Ошибка подключения MOEX: {result.stderr}"
        )

    if not result.stdout.strip():
        raise ValueError("MOEX вернула пустой ответ")

    raw = base64.b64decode(
        result.stdout.strip(),
        validate=True
    )

    return json.loads(raw.decode("utf-8-sig"))


# =====================================================
# 4. ПРЕОБРАЗОВАНИЕ ОТВЕТА MOEX
# =====================================================

def moex_table(data, block):

    section = data.get(block, {})

    return pd.DataFrame(
        section.get("data", []),
        columns=section.get("columns", [])
    )


# =====================================================
# 5. УНИВЕРСАЛЬНАЯ ПОСТРАНИЧНАЯ ЗАГРУЗКА
# =====================================================

def load_history_pages(url, params):

    start = 0

    for page in range(1000):

        request_params = {
            **params,
            "start": start,
            "iss.meta": "off",
            "iss.only": "history,history.cursor"
        }

        data = download_moex_json(
            url + "?" + urlencode(request_params)
        )

        if "history" not in data:
            raise ValueError(
                "В ответе MOEX отсутствует блок history"
            )

        df = moex_table(data, "history")

        if df.empty:
            break

        yield df

        start += len(df)

        cursor = moex_table(
            data, "history.cursor"
        )

        if not cursor.empty and "TOTAL" in cursor.columns:

            total = int(cursor.iloc[0]["TOTAL"])

            if start >= total:
                break

    else:
        raise RuntimeError(
            "Превышен лимит страниц MOEX"
        )

# 6. ПАКЕТНАЯ ЗАГРУЗКА КОТИРОВОК НА ДАТУ
# =====================================================

def download_board_history(board, market, date):

    url = (
        "https://iss.moex.com/iss/history/"
        "engines/stock/"
        f"markets/{market}/"
        f"boards/{board}/securities.json"
    )

    columns = [
        "TRADEDATE",
        "SECID",
        "BOARDID",
        "LEGALCLOSEPRICE"
    ]

    params = {
        "date": date,
        "history.columns": ",".join(columns)
    }

    parts = []

    for page, df in enumerate(
        load_history_pages(url, params)
    ):

        if not set(columns).issubset(df.columns):
            raise ValueError(
                f"{board}: отсутствуют необходимые поля. "
                f"Получены: {df.columns.tolist()}"
            )

        df["TRADEDATE"] = pd.to_datetime(
            df["TRADEDATE"],
            errors="coerce"
        ).dt.strftime("%Y-%m-%d")

        # Проверяем даты в ответе
        actual_dates = df["TRADEDATE"].dropna().unique()

        wrong_dates = [
            d for d in actual_dates if d != date
        ]

        if wrong_dates:

            print(
                f"{board}: получены записи "
                f"за другие даты: {wrong_dates[:5]}"
            )

            # Если API не применил фильтр даты,
            # не загружаем всю историю биржи.
            # Отсутствующие цены будут найдены
            # отдельными запросами по инструментам.
            if page == 0 and date not in actual_dates:
                print(
                    f"{board}: пакетный ответ не содержит "
                    f"{date}; используем резервный поиск"
                )
                break

        # Берём строго нужную дату
        filtered = df[
            df["TRADEDATE"] == date
        ].copy()

        if not filtered.empty:
            parts.append(filtered[columns])

    if not parts:

        print(f"{board}: котировки на {date} не найдены")

        return pd.DataFrame(columns=columns)

    result = pd.concat(
        parts, ignore_index=True
    )

    print(
        f"{board}: загружено {len(result)} записей"
    )

    return result


# =====================================================
# 7. ЗАГРУЗКА ОБЩЕЙ ТАБЛИЦЫ
# =====================================================

def download_all_moex(date):

    print(f"\nЗагрузка MOEX на {date}")

    with ThreadPoolExecutor(max_workers=3) as executor:

        futures = {
            board: executor.submit(
                download_board_history,
                board,
                market,
                date
            )
            for board, market in BOARDS.items()
        }

        frames = [
            future.result()
            for future in futures.values()
        ]

    moex_df = pd.concat(
        frames, ignore_index=True
    )

    moex_df["SECID"] = (
        moex_df["SECID"]
        .astype("string")
        .str.upper()
        .str.strip()
    )

    moex_df["BOARDID"] = (
        moex_df["BOARDID"]
        .astype("string")
        .str.upper()
        .str.strip()
    )

    moex_df["LEGALCLOSEPRICE"] = pd.to_numeric(
        moex_df["LEGALCLOSEPRICE"],
        errors="coerce"
    )

    moex_df.loc[
        moex_df["LEGALCLOSEPRICE"] <= 0,
        "LEGALCLOSEPRICE"
    ] = np.nan

    # Не допускаем неоднозначного объединения
    if moex_df.duplicated(
        ["SECID", "BOARDID"]
    ).any():

        raise ValueError(
            "Обнаружены дубликаты SECID + BOARDID"
        )

    moex_df = moex_df.rename(
        columns={"LEGALCLOSEPRICE": "close"}
    )

    print(
        f"\nВсего котировок MOEX: {len(moex_df)}"
    )

    return moex_df

# 8. ОПРЕДЕЛЕНИЕ РЕЖИМА ТОРГОВ
# =====================================================

def assign_board(security_type, issuer):

    security_type = str(security_type).lower()
    issuer = str(issuer).upper()

    if "акци" in security_type:
        return "TQBR"

    if "облигац" in security_type:

        if "МИНФИН РОССИИ" in issuer:
            return "TQOB"

        return "TQCB"

    return None


# =====================================================
# 9. ПОИСК SECID ПО ISIN
# =====================================================

def find_secid(isin, board, available_pairs):

    url = (
        "https://iss.moex.com/iss/securities.json?"
        + urlencode({
            "q": isin,
            "iss.meta": "off",
            "iss.only": "securities"
        })
    )

    data = download_moex_json(url)

    df = moex_table(data, "securities")

    if df.empty:
        return None

    df.columns = df.columns.str.upper()

    if "SECID" not in df.columns:
        return None

    isin_mask = pd.Series(
        False, index=df.index
    )

    if "ISIN" in df.columns:

        isin_mask = (
            df["ISIN"]
            .astype(str)
            .str.upper() == isin
        )

    secid_mask = (
        df["SECID"]
        .astype(str)
        .str.upper() == isin
    )

    df = df[isin_mask | secid_mask].copy()

    if df.empty:
        return None

    # Предпочитаем нужный основной режим
    if "PRIMARY_BOARDID" in df.columns:

        preferred = df[
            df["PRIMARY_BOARDID"]
            .astype(str)
            .str.upper() == board
        ]

        if not preferred.empty:
            df = preferred

    secids = (
        df["SECID"]
        .dropna()
        .astype(str)
        .str.upper()
        .unique()
        .tolist()
    )

    # Предпочитаем SECID из котировок на дату
    available = [
        s for s in secids
        if (s, board) in available_pairs
    ]

    if len(available) == 1:
        return available[0]

    if len(available) > 1:
        raise ValueError(
            f"Несколько SECID для {isin}: {available}"
        )

    if len(secids) == 1:
        return secids[0]

    if len(secids) > 1:
        raise ValueError(
            f"Неоднозначный ISIN {isin}: {secids}"
        )

    return None

# 10. ПОИСК ПОСЛЕДНЕЙ ИЗВЕСТНОЙ ЦЕНЫ
# =====================================================

def get_last_known_close(secid, board, report_date):

    market = BOARDS[board]

    url = (
        "https://iss.moex.com/iss/history/"
        "engines/stock/"
        f"markets/{market}/"
        f"boards/{board}/"
        f"securities/{quote(str(secid), safe='')}.json"
    )

    target = pd.Timestamp(report_date).normalize()

    # Включаем отчётную дату
    end_date = target

    earliest = target - pd.Timedelta(
        days=MAX_LOOKBACK_DAYS
    )

    for window in SEARCH_WINDOWS:

        start_date = max(
            earliest,
            end_date - pd.Timedelta(days=window - 1)
        )

        params = {
            "from": start_date.strftime("%Y-%m-%d"),
            "till": end_date.strftime("%Y-%m-%d"),
            "history.columns": (
                "TRADEDATE,LEGALCLOSEPRICE"
            )
        }

        parts = []

        for df in load_history_pages(url, params):

            required = {
                "TRADEDATE",
                "LEGALCLOSEPRICE"
            }

            if not required.issubset(df.columns):
                raise ValueError(
                    f"{secid}: отсутствуют поля истории"
                )

            parts.append(df)

        if parts:

            history_df = pd.concat(
                parts, ignore_index=True
            )

            history_df["TRADEDATE"] = pd.to_datetime(
                history_df["TRADEDATE"],
                errors="coerce"
            )

            history_df["LEGALCLOSEPRICE"] = pd.to_numeric(
                history_df["LEGALCLOSEPRICE"],
                errors="coerce"
            )

            # Только действительные цены в прошлом
            history_df = history_df[
                history_df["TRADEDATE"].between(
                    start_date, end_date
                )
                & history_df["LEGALCLOSEPRICE"].notna()
                & (history_df["LEGALCLOSEPRICE"] > 0)
            ]

            if not history_df.empty:

                last = history_df.sort_values(
                    "TRADEDATE",
                    ascending=False
                ).iloc[0]

                return (
                    float(last["LEGALCLOSEPRICE"]),
                    last["TRADEDATE"]
                )

        # Переходим к более раннему интервалу
        if start_date <= earliest:
            break

        end_date = (
            start_date - pd.Timedelta(days=1)
        )

    return np.nan, pd.NaT

# 11. ПОДГОТОВКА ПОРТФЕЛЯ
# =====================================================

def prepare_portfolio(fin_inv):

    required = [
        "ISIN",
        "Тип ФВ",
        "Наименование эмитента"
    ]

    portfolio = fin_inv.copy()

    portfolio.columns = (
        portfolio.columns.astype(str).str.strip()
    )

    missing = [
        col for col in required
        if col not in portfolio.columns
    ]

    if missing:
        raise ValueError(
            f"Отсутствуют колонки: {missing}"
        )

    portfolio = portfolio[required].copy()

    portfolio["ISIN"] = (
        portfolio["ISIN"]
        .astype("string")
        .str.strip()
        .str.upper()
    )

    portfolio = portfolio.drop_duplicates(
        subset="ISIN"
    ).reset_index(drop=True)

    portfolio["BOARDID"] = portfolio.apply(
        lambda row: assign_board(
            row["Тип ФВ"],
            row["Наименование эмитента"]
        ),
        axis=1
    )

    return portfolio


# =====================================================
# 12. СОПОСТАВЛЕНИЕ ISIN -> SECID
# =====================================================

def prepare_isin_map(portfolio, moex_df):

    available_pairs = set(
        zip(
            moex_df["SECID"],
            moex_df["BOARDID"]
        )
    )

    mapping = {}
    errors = {}
    tasks = []

    for _, row in portfolio.iterrows():

        isin = row["ISIN"]
        board = row["BOARDID"]

        if pd.isna(isin) or pd.isna(board):
            continue

        key = (isin, board)

        # Для облигаций SECID часто равен ISIN
        if (
            board in ("TQOB", "TQCB")
            and key in available_pairs
        ):
            mapping[key] = isin

        else:
            tasks.append(key)

    tasks = list(dict.fromkeys(tasks))

    print(
        f"\nПоиск SECID: {len(tasks)} бумаг"
    )

    def resolve(task):

        isin, board = task

        try:

            secid = find_secid(
                isin,
                board,
                available_pairs
            )

            # Для облигаций дополнительно
            # допускаем SECID = ISIN
            if secid is None and board in ("TQOB", "TQCB"):
                secid = isin

            return task, secid, None

        except Exception as e:

            return task, None, str(e)

    with ThreadPoolExecutor(
        max_workers=MAX_WORKERS
    ) as executor:

        results = list(
            executor.map(resolve, tasks)
        )

    for key, secid, error in results:

        if error is not None:
            errors[key] = error

        elif secid is not None:
            mapping[key] = secid

    return mapping, errors


# =====================================================
# 13. ОБЪЕДИНЕНИЕ С FIN_INV
# =====================================================

def merge_with_portfolio(fin_inv, moex_df, report_date):

    portfolio = prepare_portfolio(fin_inv)

    mapping, errors = prepare_isin_map(
        portfolio, moex_df
    )

    portfolio["SECID"] = portfolio.apply(
        lambda row: mapping.get(
            (row["ISIN"], row["BOARDID"])
        ),
        axis=1
    )

    merged = portfolio.merge(
        moex_df[
            ["SECID", "BOARDID", "close"]
        ],
        on=["SECID", "BOARDID"],
        how="left",
        validate="many_to_one"
    )

    merged["price_date"] = pd.NaT

    merged["lag_days"] = pd.Series(
        pd.NA,
        index=merged.index,
        dtype="Int64"
    )

    merged["status"] = "Нет котировки"

    target = pd.Timestamp(report_date)

    # Найденные цены на отчётную дату
    ok = merged["close"].notna()

    merged.loc[ok, "price_date"] = target
    merged.loc[ok, "lag_days"] = 0
    merged.loc[ok, "status"] = "OK"

    # Статусы для остальных
    for idx, row in merged.loc[~ok].iterrows():

        key = (row["ISIN"], row["BOARDID"])

        if pd.isna(row["ISIN"]):
            status = "Пустой ISIN"

        elif pd.isna(row["BOARDID"]):
            status = "Неизвестный тип бумаги"

        elif key in errors:
            status = f"Ошибка SECID: {errors[key]}"

        elif pd.isna(row["SECID"]):
            status = "SECID не найден"

        else:
            status = "Требуется поиск истории"

        merged.at[idx, "status"] = status

    return merged


# =====================================================
# 14. ЗАПОЛНЕНИЕ ОТСУТСТВУЮЩИХ ЦЕН
# =====================================================

def fill_missing_closes(merged, report_date):

    merged = merged.copy()

    target = pd.Timestamp(report_date)

    tasks = []

    for idx, row in merged.iterrows():

        if (
            pd.isna(row["close"])
            and pd.notna(row["SECID"])
            and row["BOARDID"] in BOARDS
        ):

            tasks.append(
                (idx, row["SECID"], row["BOARDID"])
            )

    print(
        f"\nПоиск последних известных цен: {len(tasks)}"
    )

    def process(task):

        idx, secid, board = task

        try:

            price, price_date = get_last_known_close(
                secid,
                board,
                report_date
            )

            return idx, price, price_date, None

        except Exception as e:

            return idx, np.nan, pd.NaT, str(e)

    with ThreadPoolExecutor(
        max_workers=MAX_WORKERS
    ) as executor:

        results = list(
            executor.map(process, tasks)
        )

    for idx, price, price_date, error in results:

        if pd.notna(price):

            merged.at[idx, "close"] = price
            merged.at[idx, "price_date"] = price_date

            lag = (target - price_date).days

            merged.at[idx, "lag_days"] = lag

            if lag == 0:
                merged.at[idx, "status"] = (
                    "OK (индивидуальный запрос)"
                )
            else:
                merged.at[idx, "status"] = (
                    "OK (последняя известная цена)"
                )

        elif error is not None:

            merged.at[idx, "status"] = (
                f"Ошибка поиска истории: {error}"
            )

        else:

            merged.at[idx, "status"] = (
                "Историческая цена не найдена"
            )

    return merged

# 15. ОСНОВНАЯ ФУНКЦИЯ ПАРСЕРА
# =====================================================

def parse_fin_inv(fin_inv, report_date):

    # 1. Пакетная загрузка котировок
    moex_df = download_all_moex(report_date)

    # 2. Сопоставление с портфелем
    merged = merge_with_portfolio(
        fin_inv,
        moex_df,
        report_date
    )

    # 3. Поиск последних известных цен
    merged = fill_missing_closes(
        merged,
        report_date
    )

    # 4. Итоговый DataFrame
    result_df = merged[
        [
            "ISIN",
            "Наименование эмитента",
            "close"
        ]
    ].copy()

    # 5. Диагностика
    status_df = merged[
        [
            "ISIN",
            "SECID",
            "BOARDID",
            "price_date",
            "lag_days",
            "status"
        ]
    ].copy()

    return result_df, status_df, moex_df
