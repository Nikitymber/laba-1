"""Кривая бескупонной доходности государственных облигаций Банка России.

Установка:
    python -m pip install pandas lxml

Использование:
    from cbr_kbd_parser import parse_cbr_kbd
    df = parse_cbr_kbd("2026-10-13")
    latest = parse_cbr_kbd()
    history = parse_cbr_kbd(start="2026-10-01", end="2026-10-13")

Запуск:
    python cbr_kbd_parser.py --date 2026-10-13 --output cbr_kbd.csv
    python cbr_kbd_parser.py --start 2026-10-01 --end 2026-10-13

Источник: https://www.cbr.ru/hd_base/zcyc_params/
Индекс результата — Date (DatetimeIndex), сроки — столбцы 3M, 6M, 9M,
1Y, 2Y, 3Y, 5Y, 7Y, 10Y, 15Y, 20Y, 30Y. Ставки в процентах годовых:
16.57 означает 16.57%, а не 0.1657. Пропущенные значения — NaN.
При отсутствии публикаций за выбранный период возникает ValueError.
Загрузка: Windows PowerShell, системный HTTPS-прокси, учётные данные
текущего пользователя Windows; передача HTML через Base64 без файлов.
"""

import argparse
import base64
import binascii
import os
import re
import subprocess
import urllib.request
from io import StringIO
from pathlib import Path
from typing import Optional
from urllib.parse import urlencode

import pandas as pd


BASE_URL = "https://www.cbr.ru/hd_base/zcyc_params/"
MATURITY_YEARS = {
    "3M": 0.25, "6M": 0.5, "9M": 0.75,
    "1Y": 1.0, "2Y": 2.0, "3Y": 3.0, "5Y": 5.0, "7Y": 7.0,
    "10Y": 10.0, "15Y": 15.0, "20Y": 20.0, "30Y": 30.0,
}
MATURITIES = list(MATURITY_YEARS)


PS_SCRIPT = r"""
$ErrorActionPreference = "Stop";
$ProgressPreference = "SilentlyContinue";

[System.Net.ServicePointManager]::SecurityProtocol = (
    [System.Net.ServicePointManager]::SecurityProtocol -bor
    [System.Net.SecurityProtocolType]::Tls12
);

$proxy = [System.Net.WebProxy]::new($env:CBR_KBD_PROXY);
$proxy.Credentials = [System.Net.CredentialCache]::DefaultNetworkCredentials;

$client = [System.Net.WebClient]::new();
$client.Proxy = $proxy;
$client.Headers["User-Agent"] = "Mozilla/5.0";
$client.Headers["Accept"] = "text/html";

try {
    $bytes = $client.DownloadData($env:CBR_KBD_URL);
    [Console]::Out.Write([Convert]::ToBase64String($bytes));
}
catch {
    [Console]::Error.WriteLine($_.Exception.Message);
    exit 1;
}
finally {
    $client.Dispose();
}
"""


def download_cbr_html(
    url: str,
    proxy_url: Optional[str] = None,
    timeout: float = 90,
) -> str:
    """Загрузить HTML через корпоративный HTTPS-прокси с авторизацией Windows."""
    if timeout <= 0:
        raise ValueError("timeout должен быть больше нуля.")
    if proxy_url is None:
        proxy_url = urllib.request.getproxies().get("https")
    if not proxy_url:
        raise RuntimeError("Корпоративный HTTPS-прокси не найден.")

    env = os.environ.copy()
    env["CBR_KBD_PROXY"] = proxy_url
    env["CBR_KBD_URL"] = url

    try:
        result = subprocess.run(
            [
                "powershell.exe", "-NoProfile", "-NonInteractive",
                "-Command", PS_SCRIPT,
            ],
            capture_output=True,
            text=True,
            errors="replace",
            env=env,
            timeout=timeout,
        )
    except FileNotFoundError as exc:
        raise RuntimeError(
            "powershell.exe не найден. Загрузка рассчитана на Windows PowerShell."
        ) from exc
    except subprocess.TimeoutExpired as exc:
        raise TimeoutError(
            f"Превышено время загрузки КБД Банка России через прокси ({timeout} с)."
        ) from exc

    if result.returncode != 0:
        detail = result.stderr.strip() or f"код завершения {result.returncode}"
        raise RuntimeError(f"Ошибка подключения к Банку России через прокси: {detail}")

    encoded = result.stdout.strip()
    if not encoded:
        raise ValueError("Банк России вернул пустой ответ.")
    try:
        raw = base64.b64decode(encoded, validate=True)
    except (binascii.Error, ValueError) as exc:
        raise ValueError("PowerShell вернул некорректный Base64.") from exc

    if not raw:
        raise ValueError("Банк России вернул пустой HTML.")
    return raw.decode("utf-8-sig")


def _parse_html(html: str) -> pd.DataFrame:
    """Разобрать двухстрочный заголовок и ставки с десятичной запятой."""
    try:
        tables = pd.read_html(
            StringIO(html),
            flavor="lxml",
            decimal=",",
            thousands=None,
            na_values=["N/A", "n/a", "—", "-", "--"],
        )
    except ValueError as exc:
        raise ValueError(
            "Таблица КБД не найдена. Проверьте ответ Банка России или прокси."
        ) from exc

    for table in tables:
        # В MultiIndex первый уровень — «Срок до погашения, лет», второй — срок.
        columns = [str(c[-1] if isinstance(c, tuple) else c).strip() for c in table.columns]
        if "Дата" in columns:
            table.columns = columns
            break
    else:
        raise ValueError("Не найдена таблица КБД с колонкой «Дата».")

    if table.empty:
        raise ValueError("Банк России не опубликовал КБД за запрошенный период.")
    if table.columns.has_duplicates:
        raise ValueError("В таблице КБД обнаружены повторяющиеся столбцы.")

    by_years = {years: label for label, years in MATURITY_YEARS.items()}
    rename = {"Дата": "Date"}
    for column in table.columns:
        if column == "Дата":
            continue
        normalized = re.sub(r"\s+", "", column).replace(",", ".")
        try:
            years = float(normalized)
        except ValueError:
            continue
        if years in by_years:
            rename[column] = by_years[years]

    missing = [label for label in MATURITIES if label not in rename.values()]
    if missing:
        raise ValueError(f"В таблице КБД отсутствуют сроки: {missing}.")

    table = table.rename(columns=rename)
    if table.columns.has_duplicates:
        raise ValueError("Несколько столбцов КБД соответствуют одному сроку.")
    df = table[["Date", *MATURITIES]].copy()
    df["Date"] = pd.to_datetime(df["Date"], format="%d.%m.%Y", errors="raise")

    for column in MATURITIES:
        values = (
            df[column].astype("string")
            .str.replace(r"\s+", "", regex=True)
            .str.replace(",", ".", regex=False)
        )
        df[column] = pd.to_numeric(values, errors="raise").astype(float)

    df = df.set_index("Date").sort_index()
    if df.index.has_duplicates:
        raise ValueError("В таблице КБД обнаружены повторяющиеся даты.")
    if df.isna().all(axis=1).any():
        raise ValueError("В таблице КБД есть дата без единой ставки.")
    return df


def _date(value: str) -> pd.Timestamp:
    """Привести дату YYYY-MM-DD к календарной дате без часового пояса."""
    timestamp = pd.Timestamp(value)
    if pd.isna(timestamp):
        raise ValueError("Не указана корректная дата.")
    return pd.Timestamp(timestamp.date())


def parse_cbr_kbd(
    date: Optional[str] = None,
    *,
    start: Optional[str] = None,
    end: Optional[str] = None,
    proxy_url: Optional[str] = None,
    timeout: float = 90,
) -> pd.DataFrame:
    """Получить КБД на дату, за диапазон включительно или последнюю публикацию.

    parse_cbr_kbd("2026-10-13"): ровно выбранная дата.
    parse_cbr_kbd(start="2026-10-01", end="2026-10-13"): опубликованные даты периода.
    parse_cbr_kbd(): одна последняя опубликованная кривая с основной страницы.
    Если за выбранную дату/период нет данных, возникает ValueError.
    """
    if date is not None and (start is not None or end is not None):
        raise ValueError("Укажите либо date, либо диапазон start и end.")

    if date is not None:
        lower = upper = _date(date)
    elif start is not None or end is not None:
        if start is None or end is None:
            raise ValueError("Для диапазона нужно указать и start, и end.")
        lower, upper = _date(start), _date(end)
        if lower > upper:
            raise ValueError("start не может быть позже end.")
    else:
        lower = upper = None

    url = BASE_URL
    if lower is not None:
        url += "?" + urlencode({
            "UniDbQuery.Posted": "True",
            "UniDbQuery.From": lower.strftime("%d.%m.%Y"),
            "UniDbQuery.To": upper.strftime("%d.%m.%Y"),
        })

    html = download_cbr_html(url, proxy_url=proxy_url, timeout=timeout)
    df = _parse_html(html)

    if lower is None:
        return df.tail(1).copy()
    if not ((df.index >= lower) & (df.index <= upper)).all():
        raise ValueError("Банк России вернул даты за пределами запрошенного периода.")
    return df


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--date", help="Дата YYYY-MM-DD")
    parser.add_argument("--start", help="Начало диапазона YYYY-MM-DD")
    parser.add_argument("--end", help="Конец диапазона YYYY-MM-DD")
    parser.add_argument("--output", type=Path, help="Путь для сохранения CSV")
    parser.add_argument("--timeout", type=float, default=90, help="Тайм-аут в секундах")
    parser.add_argument("--proxy", help="Адрес прокси, если его нужно задать явно")
    args = parser.parse_args()

    data = parse_cbr_kbd(
        args.date, start=args.start, end=args.end,
        proxy_url=args.proxy, timeout=args.timeout,
    )
    suffix = f"{data.index[0]:%Y-%m-%d}"
    if len(data) > 1:
        suffix += f"_{data.index[-1]:%Y-%m-%d}"
    output = args.output or Path(f"cbr_kbd_{suffix}.csv")
    output.parent.mkdir(parents=True, exist_ok=True)
    data.to_csv(output, encoding="utf-8-sig", date_format="%Y-%m-%d")
    print(data.to_string())
    print(f"\nСохранено {len(data)} дат КБД в {output}")
